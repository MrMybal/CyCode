cmake_minimum_required(VERSION 3.24)
project(CyCodeManager LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The Manager desktop host currently supports Windows only")
endif()
get_filename_component(CYCODE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CYCODE_COMMON_DEPS "${CYCODE_ROOT}/runtime/deps" CACHE PATH "Core source dependencies")
include("${CYCODE_ROOT}/cmake/CyCodeComponents.cmake")
cycode_add_components(Process)
set(CYUI_ROOT "${CYCODE_ROOT}/runtime/deps/manager-cyui" CACHE PATH "Pinned CyUI source")
file(READ "${CMAKE_CURRENT_LIST_DIR}/cyui-source.lock.json" _lock)
string(JSON _revision GET "${_lock}" revision)
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYUI_ROOT}" -C "${CYUI_ROOT}" rev-parse HEAD OUTPUT_VARIABLE _actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT _actual STREQUAL _revision)
  message(FATAL_ERROR "CyUI source does not match CyCodeManager/cyui-source.lock.json")
endif()
set(CYUI_IMGUI_DIR "${CYUI_ROOT}/third_party/imgui")
add_library(manager_imgui STATIC "${CYUI_IMGUI_DIR}/imgui.cpp" "${CYUI_IMGUI_DIR}/imgui_draw.cpp"
  "${CYUI_IMGUI_DIR}/imgui_tables.cpp" "${CYUI_IMGUI_DIR}/imgui_widgets.cpp"
  "${CYUI_IMGUI_DIR}/backends/imgui_impl_win32.cpp" "${CYUI_IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
target_include_directories(manager_imgui PUBLIC "${CYUI_IMGUI_DIR}" "${CYUI_IMGUI_DIR}/backends")
target_compile_definitions(manager_imgui PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
set(CYUI_IMGUI_TARGET manager_imgui)
set(CYUI_MODULES "Core" CACHE STRING "Manager modules" FORCE)
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CYUI_ROOT}" "${CMAKE_BINARY_DIR}/cyui")
add_executable(CyCodeManager WIN32 Manager.cpp ManagerLocale.h)
target_compile_features(CyCodeManager PRIVATE cxx_std_20)
target_link_libraries(CyCodeManager PRIVATE CyUI::Core CyCode::Process d3d11 dxgi d3dcompiler shell32 ole32)
target_compile_definitions(CyCodeManager PRIVATE _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
if(MSVC)
  target_compile_options(CyCodeManager PRIVATE /W4 /utf-8 /EHsc /permissive-)
endif()
