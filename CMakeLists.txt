cmake_minimum_required(VERSION 3.24)
project(CyCodeCore VERSION 0.2.3 LANGUAGES C CXX)
option(BUILD_TESTING "Build explicit native tests; never run implicitly" OFF)
include(CTest)
include(cmake/CyCodeComponents.cmake)
if(CYCODE_BUILD_GUI)
  message(FATAL_ERROR "Build the graphical host from https://github.com/MrMybal/CyCodeIDE")
endif()
set(CYCODE_COMPONENTS "Core" CACHE STRING "Native components: Core or Files;Snippets;Protocol;Documents;TextSession;TextBridge;Process;Transport")
if(NOT CYCODE_COMPONENTS)
  message(FATAL_ERROR "Select at least one CyCode component")
endif()
if(Core IN_LIST CYCODE_COMPONENTS)
  # The full project Engine remains the default and owns its existing services.
  add_subdirectory(app)
  set(_cycode_extra ${CYCODE_COMPONENTS})
  list(REMOVE_ITEM _cycode_extra Core)
  if(_cycode_extra)
    cycode_add_components(${_cycode_extra})
  endif()
else()
  cycode_add_components(${CYCODE_COMPONENTS})
endif()
add_custom_target(cycode_components ALL)
foreach(_cycode_component IN LISTS CYCODE_COMPONENTS)
  get_target_property(_cycode_target CyCode::${_cycode_component} ALIASED_TARGET)
  add_dependencies(cycode_components ${_cycode_target})
endforeach()
