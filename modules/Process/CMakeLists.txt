if(NOT TARGET CyCode::Files)
  message(FATAL_ERROR "Add CyCode/common before Process")
endif()
find_package(Threads REQUIRED)
add_library(cycode_process STATIC EXCLUDE_FROM_ALL src/Process.cpp src/ProcessText.cpp src/Tools.cpp)
target_include_directories(cycode_process PUBLIC include)
target_link_libraries(cycode_process PUBLIC CyCode::Files Threads::Threads)
set_target_properties(cycode_process PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_compile_definitions(cycode_process PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
add_library(CyCode::Process ALIAS cycode_process)
