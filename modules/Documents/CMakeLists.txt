# Standalone native text buffers and optional session/plugin ownership.
if(NOT TARGET CyCode::Files)
  message(FATAL_ERROR "Add CyCode/common before Documents")
endif()
find_package(Threads REQUIRED)
add_library(cycode_text_contracts INTERFACE)
target_include_directories(cycode_text_contracts INTERFACE include)
target_link_libraries(cycode_text_contracts INTERFACE CyCode::Files)
add_library(CyCode::TextContracts ALIAS cycode_text_contracts)
add_library(cycode_documents STATIC EXCLUDE_FROM_ALL src/Document.cpp src/Encoding.cpp src/Search.cpp src/DocumentState.cpp)
target_link_libraries(cycode_documents PUBLIC CyCode::TextContracts)
set_target_properties(cycode_documents PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(CyCode::Documents ALIAS cycode_documents)
add_library(cycode_text_session STATIC EXCLUDE_FROM_ALL src/DocumentSession.cpp src/TextPlugins.cpp)
target_link_libraries(cycode_text_session PUBLIC CyCode::Documents Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(cycode_text_session PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(CyCode::TextSession ALIAS cycode_text_session)
