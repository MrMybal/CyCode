# Native source API 1. Can be embedded without configuring the Engine or CLI.
if(TARGET cycode_files OR TARGET cycode_snippets)
  message(FATAL_ERROR "CyCode common must be configured once; reuse the existing targets")
endif()
if(NOT DEFINED CYCODE_COMMON_DEPS)
  get_filename_component(CYCODE_COMMON_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/deps" ABSOLUTE)
endif()
if(NOT EXISTS "${CYCODE_COMMON_DEPS}/json/single_include/nlohmann/json.hpp")
  message(FATAL_ERROR "nlohmann/json missing: fetch CyCode native dependencies or supply CYCODE_COMMON_DEPS")
endif()
add_library(cycode_files STATIC src/Paths.cpp src/TextFile.cpp)
target_include_directories(cycode_files PUBLIC include "${CYCODE_COMMON_DEPS}/json/single_include")
target_compile_features(cycode_files PUBLIC cxx_std_20)
set_target_properties(cycode_files PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_compile_definitions(cycode_files PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(cycode_files PUBLIC /utf-8 /permissive- /Zc:__cplusplus /EHsc)
endif()
add_library(cycode_snippets STATIC EXCLUDE_FROM_ALL src/Snippet.cpp)
target_link_libraries(cycode_snippets PUBLIC cycode_files)
set_target_properties(cycode_snippets PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(CyCode::Files ALIAS cycode_files)
add_library(CyCode::Snippets ALIAS cycode_snippets)

add_library(cycode_protocol STATIC EXCLUDE_FROM_ALL src/ServiceProtocol.cpp)
target_link_libraries(cycode_protocol PUBLIC CyCode::Files)
set_target_properties(cycode_protocol PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(CyCode::Protocol ALIAS cycode_protocol)
