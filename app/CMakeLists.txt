cmake_minimum_required(VERSION 3.24)
project(CyCode VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" /utf-8)
endif()
get_filename_component(CYCODE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CYCODE_DEPS "${CYCODE_ROOT}/runtime/deps" CACHE PATH "Downloaded dependency sources")
set(CYUI_DIR "${CYCODE_ROOT}/../CyImgui_Interface" CACHE PATH "CyUI source checkout")
option(CYCODE_BUILD_GUI "Build the desktop interface" ON)
include(CTest)
find_package(Threads REQUIRED)
file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS "${CYCODE_DEPS}/vterm/src/*.c")
add_library(cycode_vterm STATIC ${VTERM_SOURCES})
target_include_directories(cycode_vterm PUBLIC "${CYCODE_DEPS}/vterm/include")
set_target_properties(cycode_vterm PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(MSVC)
  target_compile_definitions(cycode_vterm PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_library(cycode_core STATIC src/Process.cpp src/Tools.cpp src/Commands.cpp src/TextEdits.cpp src/Semantic.cpp src/Engine.cpp src/Plugins.cpp src/Bridge.cpp src/LanguageClient.cpp src/LanguageService.cpp src/BuildSystem.cpp src/Solution.cpp src/UnrealProject.cpp)
target_include_directories(cycode_core PUBLIC include "${CYCODE_DEPS}/json/single_include")
target_sources(cycode_core PRIVATE src/Terminal.cpp src/Debugger.cpp src/DebugRuntime.cpp src/Vcs.cpp src/MsBuild.cpp src/Highlighting.cpp)
target_link_libraries(cycode_core PUBLIC Threads::Threads cycode_vterm ${CMAKE_DL_LIBS})
if(UNIX AND NOT APPLE)
  target_link_libraries(cycode_core PUBLIC util)
endif()
if(WIN32)
  target_compile_definitions(cycode_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(cycode_core PUBLIC ws2_32 bcrypt advapi32)
endif()
if(MSVC)
  target_compile_options(cycode_core PUBLIC /utf-8 /permissive- /Zc:__cplusplus)
else()
  target_compile_options(cycode_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(cycode_sample MODULE plugins/sample/sample.cpp)
target_include_directories(cycode_sample PRIVATE include "${CYCODE_DEPS}/json/single_include")
set_target_properties(cycode_sample PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CYCODE_ROOT}/runtime/plugins" RUNTIME_OUTPUT_DIRECTORY "${CYCODE_ROOT}/runtime/plugins")
foreach(configuration Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${configuration}" configuration_upper)
  set_target_properties(cycode_sample PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${configuration_upper} "${CYCODE_ROOT}/runtime/plugins" RUNTIME_OUTPUT_DIRECTORY_${configuration_upper} "${CYCODE_ROOT}/runtime/plugins")
endforeach()
configure_file(plugins/sample/plugin.json "${CYCODE_ROOT}/runtime/plugins/sample.json" COPYONLY)
if(CYCODE_BUILD_GUI)
  find_package(OpenGL REQUIRED)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CYCODE_DEPS}/SDL" "${CMAKE_BINARY_DIR}/SDL" EXCLUDE_FROM_ALL)
  set(IMGUI_DIR "${CYCODE_DEPS}/imgui")
  add_library(cycode_imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp" "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp" "${IMGUI_DIR}/imgui_demo.cpp" "${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp" "${IMGUI_DIR}/misc/cpp/imgui_stdlib.cpp")
  target_include_directories(cycode_imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends" "${IMGUI_DIR}/misc/cpp")
  target_compile_definitions(cycode_imgui PUBLIC IMGUI_USE_WCHAR32)
  target_link_libraries(cycode_imgui PUBLIC SDL3::SDL3-static OpenGL::GL)
  set(CYUI_IMGUI_TARGET cycode_imgui CACHE STRING "" FORCE)
  set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CYUI_DIR}" "${CMAKE_BINARY_DIR}/cyui")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(CYCODE_EDITOR_DIR "${CMAKE_BINARY_DIR}/texteditor")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_texteditor.py"
    "${CYCODE_DEPS}/texteditor/TextEditor.h" "${CYCODE_DEPS}/texteditor/TextEditor.cpp")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_texteditor.py"
    --source "${CYCODE_DEPS}/texteditor" --destination "${CYCODE_EDITOR_DIR}" COMMAND_ERROR_IS_FATAL ANY)
  add_library(cycode_texteditor STATIC "${CYCODE_EDITOR_DIR}/TextEditor.cpp")
  target_include_directories(cycode_texteditor PUBLIC "${CYCODE_EDITOR_DIR}")
  target_link_libraries(cycode_texteditor PUBLIC cycode_imgui)
  add_library(cycode_editor STATIC src/EditorTheme.cpp src/EditorLanguage.cpp src/EditorHighlight.cpp)
  target_include_directories(cycode_editor PUBLIC include "${CYCODE_DEPS}/json/single_include")
  target_link_libraries(cycode_editor PUBLIC CyUI::CyUI cycode_texteditor)
  if(BUILD_TESTING)
    add_executable(cycode_highlight_tests tests/highlighting.cpp)
    target_link_libraries(cycode_highlight_tests PRIVATE cycode_editor)
    add_test(NAME editor_highlighting COMMAND cycode_highlight_tests)
    set_tests_properties(editor_highlighting PROPERTIES TIMEOUT 30)
  endif()
  add_executable(CyCode src/main.cpp src/Application.cpp src/CommandUI.cpp src/SemanticUI.cpp src/TerminalUI.cpp src/DebugUI.cpp src/VcsUI.cpp src/MsBuildUI.cpp src/FileIcons.cpp src/HighlightUI.cpp)
  target_link_libraries(CyCode PRIVATE cycode_core cycode_editor)
  if(WIN32)
    set_target_properties(CyCode PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_options(CyCode PRIVATE /ENTRY:mainCRTStartup)
    target_link_options(CyCode PRIVATE "/MAP:${CMAKE_BINARY_DIR}/CyCode.map")
    target_link_libraries(CyCode PRIVATE shell32)
  endif()
  add_dependencies(CyCode cycode_sample)
endif()
add_executable(cycode_cli src/cli.cpp)
target_link_libraries(cycode_cli PRIVATE cycode_core)
if(WIN32)
  target_link_libraries(cycode_cli PRIVATE shell32)
endif()

  if(MSVC)
    target_compile_options(cycode_debug_fixture PRIVATE /Zi /Od)
    target_link_options(cycode_debug_fixture PRIVATE /DEBUG /INCREMENTAL:NO /OPT:NOICF)
  else()
    target_compile_options(cycode_debug_fixture PRIVATE -g -O0)
  endif()
  add_executable(cycode_debug_tests tests/debug_tests.cpp)
  target_link_libraries(cycode_debug_tests PRIVATE cycode_core)
  add_test(NAME debugger COMMAND cycode_debug_tests "$<TARGET_FILE:cycode_debug_fixture>" "${CYCODE_ROOT}/data/validation/debug-native")
  set_tests_properties(debugger PROPERTIES TIMEOUT 65 SKIP_RETURN_CODE 77)
  add_executable(cycode_terminal_tests tests/terminal_tests.cpp)
  target_link_libraries(cycode_terminal_tests PRIVATE cycode_core)
  add_test(NAME terminal COMMAND cycode_terminal_tests)
  set_tests_properties(terminal PROPERTIES TIMEOUT 40)
  add_executable(cycode_commands_tests tests/commands_tests.cpp)
  target_link_libraries(cycode_commands_tests PRIVATE cycode_core)
  add_test(NAME commands COMMAND cycode_commands_tests)
  add_executable(cycode_solution_tests tests/solution_tests.cpp)
  target_link_libraries(cycode_solution_tests PRIVATE cycode_core)
  add_test(NAME solution_unreal COMMAND cycode_solution_tests)
  set_tests_properties(solution_unreal PROPERTIES TIMEOUT 25)
  add_executable(cycode_build_tests tests/build_tests.cpp)
  target_link_libraries(cycode_build_tests PRIVATE cycode_core)
  add_test(NAME cmake_project COMMAND cycode_build_tests)
  set_tests_properties(cmake_project PROPERTIES TIMEOUT 90)
  add_executable(cycode_language_lifecycle_tests tests/language_lifecycle_tests.cpp)
  target_link_libraries(cycode_language_lifecycle_tests PRIVATE cycode_core)
  add_test(NAME language_lifecycle COMMAND cycode_language_lifecycle_tests)
  set_tests_properties(language_lifecycle PROPERTIES TIMEOUT 25)
  add_executable(cycode_tests tests/core_tests.cpp)
  target_link_libraries(cycode_tests PRIVATE cycode_core)
  add_dependencies(cycode_tests cycode_sample)
  add_test(NAME core COMMAND cycode_tests "$<TARGET_FILE:cycode_sample>")
  find_program(CYCODE_CLANGD NAMES clangd HINTS "C:/Program Files/LLVM/bin")
  if(CYCODE_CLANGD)
    add_executable(cycode_semantic_highlight_tests tests/semantic_highlight_tests.cpp)
    target_link_libraries(cycode_semantic_highlight_tests PRIVATE cycode_core)
    add_test(NAME semantic_highlighting COMMAND cycode_semantic_highlight_tests "${CYCODE_CLANGD}")
    set_tests_properties(semantic_highlighting PROPERTIES TIMEOUT 60)
    add_executable(cycode_semantic_tests tests/semantic_tests.cpp)
    target_link_libraries(cycode_semantic_tests PRIVATE cycode_core)
    add_test(NAME semantic COMMAND cycode_semantic_tests "${CYCODE_CLANGD}")
    set_tests_properties(semantic PROPERTIES TIMEOUT 60)
    add_executable(cycode_lsp_tests tests/lsp_tests.cpp)
    target_link_libraries(cycode_lsp_tests PRIVATE cycode_core)
    add_test(NAME lsp COMMAND cycode_lsp_tests "${CYCODE_CLANGD}")
    set_tests_properties(lsp PROPERTIES TIMEOUT 45)
  endif()
endif()
