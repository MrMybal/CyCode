# Only the native core's completion adapter uses these static libraries. Keep
# their tools, JIT, test executables and unused character widths out of the build.
function(cycode_snippet_dependencies)
  if(NOT EXISTS "${CYCODE_DEPS}/pcre2/CMakeLists.txt" OR NOT EXISTS "${CYCODE_DEPS}/utf8proc/utf8proc.c")
    message(FATAL_ERROR "Snippet transform sources missing. Run python app/scripts/fetch_dependencies.py --core-only.")
  endif()
  set(BUILD_SHARED_LIBS OFF)
  set(BUILD_STATIC_LIBS ON)
  foreach(option PCRE2_BUILD_TESTS PCRE2_BUILD_PCRE2GREP PCRE2_BUILD_PCRE2_16 PCRE2_BUILD_PCRE2_32 PCRE2_SUPPORT_JIT PCRE2_REBUILD_CHARTABLES PCRE2_SHOW_REPORT UTF8PROC_ENABLE_TESTING UTF8PROC_INSTALL)
    set(${option} OFF CACHE BOOL "CyCode minimal snippet dependencies" FORCE)
  endforeach()
  set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "CyCode UTF-8 matching" FORCE)
  set(PCRE2_SUPPORT_UNICODE ON CACHE BOOL "CyCode Unicode matching" FORCE)
  set(PCRE2_STATIC_PIC ON CACHE BOOL "CyCode static matching" FORCE)
  add_subdirectory("${CYCODE_DEPS}/pcre2" "${CMAKE_BINARY_DIR}/snippet-pcre2" EXCLUDE_FROM_ALL)
  add_subdirectory("${CYCODE_DEPS}/utf8proc" "${CMAKE_BINARY_DIR}/snippet-utf8proc" EXCLUDE_FROM_ALL)
endfunction()
cycode_snippet_dependencies()
