cmake_minimum_required(VERSION 3.24)
project(CyCode VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" /utf-8)
endif()
get_filename_component(CYCODE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CYCODE_DEPS "${CYCODE_ROOT}/runtime/deps" CACHE PATH "Downloaded dependency sources")
set(CYUI_DIR "" CACHE PATH "Optional clean CyUI checkout matching app/cyui-source.lock.json")
option(CYCODE_BUILD_GUI "Build the desktop interface" ON)

if(NOT EXISTS "${CYCODE_EDITOR_ROOT}/editor/CMakeLists.txt")
  message(FATAL_ERROR "Shared editor sources missing. Run python app/scripts/fetch_dependencies.py or set CYCODE_EDITOR_ROOT to a CyCodeReader checkout.")
endif()
file(READ "${CYCODE_EDITOR_ROOT}/editor/api-version.txt" CYCODE_EDITOR_API)
string(STRIP "${CYCODE_EDITOR_API}" CYCODE_EDITOR_API)
if(NOT CYCODE_EDITOR_API STREQUAL "1")
  message(FATAL_ERROR "Unsupported shared editor source API: ${CYCODE_EDITOR_API}")
endif()
set(CYCODE_EDITOR_DEPS "${CYCODE_DEPS}")
set(CYCODE_EDITOR_BUILD_GUI ${CYCODE_BUILD_GUI})
if(CYCODE_BUILD_GUI)
  if(NOT EXISTS "${CYCODE_EDITOR_ROOT}/editor/cyui-source.lock.json")
    message(FATAL_ERROR "This shared editor predates modular CyUI. Use app/editor-source.lock.json or a current --editor-root checkout.")
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/cyui_dependency.py"
    --destination "${CYCODE_DEPS}/cyui-ff023e6" --source "${CYUI_DIR}"
    OUTPUT_VARIABLE CYUI_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  # The host pins and validates CyUI above. The shared editor's standalone lock
  # predates Core filtering/localization, so explicitly use its host-source mode.
  # No dependency source/lock is patched, and no dirty checkout is accepted.
  set(CYCODE_CYUI_DEVELOPMENT ON)
  set(CYUI_MODULES "CodeEditor;Diff;Commands;Diagnostics;DocumentViews;Inspector;Images" CACHE STRING "CyCode UI modules" FORCE)
  set(CYCODE_EDITOR_UI_MODULES "${CYUI_MODULES}")
endif()

set_target_properties(cycode_vterm PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(MSVC)
  target_compile_definitions(cycode_vterm PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_library(cycode_core STATIC src/Process.cpp src/Tools.cpp src/Commands.cpp src/TextEdits.cpp src/Semantic.cpp src/Engine.cpp src/Plugins.cpp src/Bridge.cpp src/LanguageClient.cpp src/LanguageService.cpp src/BuildSystem.cpp src/Solution.cpp src/UnrealProject.cpp)

endif()
if(WIN32)
  target_compile_definitions(cycode_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(cycode_core PUBLIC ws2_32 bcrypt advapi32)
endif()
if(MSVC)
  target_compile_options(cycode_core PUBLIC /utf-8 /permissive- /Zc:__cplusplus)

  target_compile_options(cycode_core PRIVATE -Wall -Wextra -Wpedantic)
